# Worktint Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

## Context

Developers running multiple **git worktrees of the same repo** (e.g. parallel AI agents on different branches) can't tell the VS Codium windows apart. Worktint gives each worktree a stable, unique, theme-appropriate color. It stays fully dormant until a repo actually has **more than one worktree**, then renders two layers: an always-on **status-bar indicator** (zero files, zero git footprint) and an optional **chrome tint** (title/activity/status bar + editor-tab accent) written to the worktree's `.vscode/settings.json` and hidden from git via `.git/info/exclude`.

Approved design spec: `docs/superpowers/specs/2026-07-01-worktint-design.md`. This plan implements that spec.

**Goal:** Ship an MVP VS Codium/VS Code extension that auto-colors git worktrees with a zero-footprint status-bar indicator plus an optional chrome tint.

**Architecture:** Bottom-up. Framework-free **core** modules (`src/core/*`) hold all decision logic (detection parsing, hashing, slot assignment, palette selection, color-customization build/merge/restore, exclude-file editing) and are unit-tested with `bun test`. Thin **VS Code adapters** (`src/vscode/*`) bind that logic to editor APIs (globalState "brain", config writer, status-bar item, git-branch lookup). A **controller** orchestrates detect → gate → assign → render → cleanup. Integration tests run in a real VS Code via `@vscode/test-electron`.

**Tech Stack:** TypeScript, Bun (install + `bun build` bundler + `bun test`), `@vscode/test-electron` + `@vscode/test-cli` (mocha) for in-editor integration tests, `@types/vscode`, `@types/bun`.

## Global Constraints

- **Editor target:** VS Codium / VS Code, engine `^1.90.0`. Distributed via Open VSX (publishing is out of MVP scope).
- **Activation gate:** only act when the repo has **>1 worktree** (`<gitCommonDir>/worktrees/` non-empty). Otherwise fully dormant — no status bar item, no writes, no prompts.
- **Zero git footprint:** never modify tracked `.gitignore`; hide `.vscode/settings.json` via `.git/info/exclude`. If `.vscode/settings.json` is already **tracked**, default to status-bar-only and warn once.
- **Never clobber user colors:** read existing `workbench.colorCustomizations`, merge only Worktint keys, record prior values so reset restores exactly.
- **Color seed = worktree root path** (never the branch). Branch is display-only.
- **Settings namespace:** `worktint.*`. All chrome toggles default `true`; `worktint.chrome.enabled` default `true`; `worktint.statusBarIndicator.enabled` default `true`.
- **Bundler command:** `bun build ./src/extension.ts --outdir=dist --target=node --format=cjs --external=vscode --sourcemap=linked`. Entry `main`: `./dist/extension.js`.
- **Palette:** two curated sets (light/dark), 8 colors each, every entry `{ background, foreground, accent }`. Smart-from-theme palette is deferred (Pro).

```ts
export function fnv1a(input: string): number {
  let hash = 0x811c9dc5;
  for (let i = 0; i < input.length; i++) {
    hash ^= input.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return hash >>> 0;
}
```

```ts
import * as path from 'path';

export interface FsLike {
  existsSync(p: string): boolean;
  statSync(p: string): { isDirectory(): boolean; isFile(): boolean };
  readFileSync(p: string, enc: 'utf8'): string;
  readdirSync(p: string): string[];
}
export interface WorktreeInfo { worktreePath: string; gitCommonDir: string; isMultiWorktree: boolean }

export function parseGitFile(contents: string): string | null {
  const m = contents.match(/^gitdir:\s*(.+?)\s*$/m);
  return m ? m[1] : null;
}

function findGitEntry(startDir: string, fs: FsLike): { dir: string; gitPath: string } | null {
  let dir = startDir;
  while (true) {
    const gitPath = path.join(dir, '.git');
    if (fs.existsSync(gitPath)) return { dir, gitPath };
    const parent = path.dirname(dir);
    if (parent === dir) return null;
    dir = parent;
  }
}

export function detectWorktree(folderPath: string, fs: FsLike): WorktreeInfo | null {
  const found = findGitEntry(folderPath, fs);
  if (!found) return null;
  const { dir: worktreePath, gitPath } = found;

  let gitCommonDir: string;
  const st = fs.statSync(gitPath);
  if (st.isDirectory()) {
    gitCommonDir = gitPath;
  } else {
    const gitdir = parseGitFile(fs.readFileSync(gitPath, 'utf8'));
    if (!gitdir) return null;
    // .../worktrees/<id>  ->  strip last two segments
    gitCommonDir = path.dirname(path.dirname(gitdir));
  }

  const worktreesDir = path.join(gitCommonDir, 'worktrees');
  const entries = fs.existsSync(worktreesDir) ? fs.readdirSync(worktreesDir) : [];
  return { worktreePath, gitCommonDir, isMultiWorktree: entries.length > 0 };
}
```

```ts
import { fnv1a } from './hash';

export type Assignments = Record<string, number>;

export function preferredSlot(worktreePath: string, paletteSize: number): number {
  return fnv1a(worktreePath) % paletteSize;
}

export function assignSlot(
  worktreePath: string,
  existing: Assignments,
  paletteSize: number,
  override?: number,
): number {
  if (override !== undefined) return ((override % paletteSize) + paletteSize) % paletteSize;
  if (existing[worktreePath] !== undefined) return existing[worktreePath];

  const taken = new Set(Object.entries(existing).filter(([k]) => k !== worktreePath).map(([, v]) => v));
  let slot = preferredSlot(worktreePath, paletteSize);
  for (let i = 0; i < paletteSize; i++) {
    if (!taken.has(slot)) return slot;
    slot = (slot + 1) % paletteSize;
  }
  return preferredSlot(worktreePath, paletteSize); // all taken: accept a duplicate
}
```

```ts
import type { PaletteColor } from './palette';

export interface ChromeToggles { titleBar: boolean; activityBar: boolean; statusBar: boolean; editorTabs: boolean }

export const WORKTINT_KEYS: string[] = [
  'titleBar.activeBackground', 'titleBar.activeForeground',
  'titleBar.inactiveBackground', 'titleBar.inactiveForeground',
  'activityBar.background', 'activityBar.foreground',
  'statusBar.background', 'statusBar.foreground',
  'tab.activeBorderTop',
];

export function buildCustomizations(color: PaletteColor, t: ChromeToggles): Record<string, string> {
  const out: Record<string, string> = {};
  if (t.titleBar) {
    out['titleBar.activeBackground'] = color.background;
    out['titleBar.activeForeground'] = color.foreground;
    out['titleBar.inactiveBackground'] = color.background;
    out['titleBar.inactiveForeground'] = color.foreground;
  }
  if (t.activityBar) {
    out['activityBar.background'] = color.background;
    out['activityBar.foreground'] = color.foreground;
  }
  if (t.statusBar) {
    out['statusBar.background'] = color.background;
    out['statusBar.foreground'] = color.foreground;
  }
  if (t.editorTabs) out['tab.activeBorderTop'] = color.accent;
  return out;
}

export function mergeCustomizations(existing: Record<string, string>, ours: Record<string, string>): Record<string, string> {
  return { ...existing, ...ours };
}

export function restoreCustomizations(current: Record<string, string>, prior: Record<string, string | undefined>): Record<string, string> {
  const out = { ...current };
  for (const [k, v] of Object.entries(prior)) {
    if (v === undefined) delete out[k];
    else out[k] = v;
  }
  return out;
}
```

```ts
function lines(contents: string): string[] {
  return contents.length === 0 ? [] : contents.replace(/\n$/, '').split('\n');
}
export function hasLine(contents: string, line: string): boolean {
  return lines(contents).includes(line);
}
export function ensureLine(contents: string, line: string): string {
  if (hasLine(contents, line)) return contents;
  const arr = lines(contents);
  arr.push(line);
  return arr.join('\n') + '\n';
}
export function removeLine(contents: string, line: string): string {
  const arr = lines(contents).filter((l) => l !== line);
  return arr.length ? arr.join('\n') + '\n' : '';
}
```

```ts
interface GitAPI { repositories: Array<{ rootUri: vscode.Uri; state: { HEAD?: { name?: string }; onDidChange: vscode.Event<void> } }>; }

function getGitApi(): GitAPI | undefined {

}

export async function getBranch(worktreePath: string): Promise<string> {
  const ext = vscode.extensions.getExtension('vscode.git');
  if (ext && !ext.isActive) await ext.activate();
  const api = getGitApi();
  const repo = api?.repositories.find((r) => r.rootUri.fsPath === worktreePath);
  return repo?.state.HEAD?.name ?? path.basename(worktreePath);
}

export function onBranchChange(worktreePath: string, cb: () => void): vscode.Disposable {
  const api = getGitApi();
  const repo = api?.repositories.find((r) => r.rootUri.fsPath === worktreePath);
  return repo ? repo.state.onDidChange(cb) : new vscode.Disposable(() => {});
}
```

```ts
import * as vscode from 'vscode';
import * as fs from 'fs';
import * as path from 'path';
import { execFileSync } from 'child_process';
import { buildCustomizations, mergeCustomizations, restoreCustomizations, WORKTINT_KEYS, type ChromeToggles } from '../core/colorCustomizations';
import { ensureLine, removeLine } from '../core/excludeFile';
import type { PaletteColor } from '../core/palette';
import type { WriteRecord } from './brain';

const CUSTOMIZATIONS = 'colorCustomizations';

export function isSettingsTracked(worktreePath: string): boolean {
  try {
    execFileSync('git', ['-C', worktreePath, 'ls-files', '--error-unmatch', '.vscode/settings.json'], { stdio: 'ignore' });
    return true;
  } catch { return false; }
}

export class SettingsWriter {
  async applyChrome(color: PaletteColor, toggles: ChromeToggles, gitCommonDir: string, worktreePath: string): Promise<WriteRecord> {
    const cfg = vscode.workspace.getConfiguration('workbench');
    const current = { ...(cfg.get<Record<string, string>>(CUSTOMIZATIONS) ?? {}) };
    const prior: Record<string, string | undefined> = {};
    for (const k of WORKTINT_KEYS) prior[k] = current[k];

    const ours = buildCustomizations(color, toggles);
    await cfg.update(CUSTOMIZATIONS, mergeCustomizations(current, ours), vscode.ConfigurationTarget.Workspace);

    const addedExcludeLine = this.ensureExcluded(gitCommonDir, worktreePath);
    return { keys: Object.keys(ours), prior, addedExcludeLine };
  }

  async revertChrome(record: WriteRecord, gitCommonDir: string): Promise<void> {
    const cfg = vscode.workspace.getConfiguration('workbench');
    const current = { ...(cfg.get<Record<string, string>>(CUSTOMIZATIONS) ?? {}) };
    const restored = restoreCustomizations(current, record.prior);
    await cfg.update(CUSTOMIZATIONS, Object.keys(restored).length ? restored : undefined, vscode.ConfigurationTarget.Workspace);
    if (record.addedExcludeLine) this.removeExcluded(gitCommonDir);
  }

  private excludePath(gitCommonDir: string): string { return path.join(gitCommonDir, 'info', 'exclude'); }

  private ensureExcluded(gitCommonDir: string, _worktreePath: string): boolean {
    const p = this.excludePath(gitCommonDir);
    const prev = fs.existsSync(p) ? fs.readFileSync(p, 'utf8') : '';
    const next = ensureLine(prev, '.vscode/settings.json');
    if (next === prev) return false;
    fs.mkdirSync(path.dirname(p), { recursive: true });
    fs.writeFileSync(p, next);
    return true;
  }

  private removeExcluded(gitCommonDir: string): void {
    const p = this.excludePath(gitCommonDir);
    if (!fs.existsSync(p)) return;
    fs.writeFileSync(p, removeLine(fs.readFileSync(p, 'utf8'), '.vscode/settings.json'));
  }
}
```

```ts
import * as vscode from 'vscode';
import { Brain } from './vscode/brain';
import { Controller } from './controller';

let controller: Controller | undefined;

export function activate(context: vscode.ExtensionContext): void {
  const brain = new Brain(context.globalState);
  controller = new Controller(brain);

  const folder = vscode.workspace.workspaceFolders?.[0]?.uri.fsPath;
  if (folder) void controller.activate(folder);

  context.subscriptions.push(
    vscode.commands.registerCommand('worktint.pickColor', () => controller?.pickColor()),
    vscode.commands.registerCommand('worktint.resetWorktree', () => controller?.resetThisWorktree()),
    vscode.commands.registerCommand('worktint.resetAll', () => controller?.resetAll()),
    vscode.commands.registerCommand('worktint.toggleChrome', async () => {
      const c = vscode.workspace.getConfiguration('worktint');
      await c.update('chrome.enabled', !c.get('chrome.enabled', true), vscode.ConfigurationTarget.Global);
      if (folder) void controller?.activate(folder);
    }),
    vscode.workspace.onDidChangeConfiguration((e) => {
      if (e.affectsConfiguration('worktint') && folder) void controller?.activate(folder);
    }),
    { dispose: () => controller?.dispose() },
  );

  // Re-evaluate when worktrees are added/removed under the git common dir.
  if (folder) {
    const watcher = vscode.workspace.createFileSystemWatcher('**/.git/worktrees/**');
    watcher.onDidCreate(() => controller?.activate(folder));
    watcher.onDidDelete(() => controller?.activate(folder));
    context.subscriptions.push(watcher);
  }
}

export function deactivate(): void { controller?.dispose(); }
```
